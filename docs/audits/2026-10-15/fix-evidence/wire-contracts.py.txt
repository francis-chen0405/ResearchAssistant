from __future__ import annotations

import json
import subprocess
import tempfile
from pathlib import Path

from fastapi.testclient import TestClient

from researchassistant.api.application import create_app
from researchassistant.api.runtime import ApiRuntime
from researchassistant.api.live_service import LiveResearchController
from researchassistant.api.service_manager import WigoloServiceManager
from researchassistant.contracts.models import ValidationError, ValidationErrorCode

ROOT = Path.cwd()


def compile_example(source: str, directory: Path, case: str) -> dict[str, object]:
    path = directory / f'{case}.ts'
    path.write_text(
        f'import type {{ ApiSchemas, ApiResponses }} from {json.dumps(str(ROOT / "frontend/lib/generated/api-contracts"))};\n'
        + source + '\n'
    )
    result = subprocess.run(
        ['node', str(ROOT / 'frontend/node_modules/typescript/bin/tsc'),
         '--noEmit', '--strict', '--skipLibCheck', '--target', 'ES2022',
         '--module', 'ESNext', '--moduleResolution', 'bundler',
         '--typeRoots', str(ROOT / 'frontend/node_modules/@types'), str(path)],
        text=True, capture_output=True, check=False,
    )
    return {'case': case, 'exit_code': result.returncode, 'diagnostic': result.stdout + result.stderr}


def main() -> None:
    runtime = ApiRuntime(controller=LiveResearchController(environment={}), services=WigoloServiceManager(base_environment={}), environment={})
    client = TestClient(create_app(runtime, load_keychain_on_start=False, allowed_hosts=("testserver",)))
    response = client.get("/api/model-options")
    assert response.status_code == 200
    catalog = response.json()
    error = ValidationError(
        code=ValidationErrorCode.SCHEMA_ERROR, location='fixture', message='Generated example'
    ).model_dump(mode='json')
    with tempfile.TemporaryDirectory(prefix='ra-wire-review-') as temporary:
        directory = Path(temporary)
        results = [
            compile_example(
                'const response = ' + json.dumps(catalog) + ' satisfies ApiResponses["GET /api/model-options"];',
                directory, 'real_catalog',
            ),
            compile_example(
                'const error = ' + json.dumps(error) + ' satisfies ApiSchemas["V2ReleaseValidation"]["errors"][number];',
                directory, 'real_research_error',
            ),
            compile_example(
                'const error = {loc: ["body"], msg: "Input error", type: "missing"} satisfies ApiSchemas["V2ReleaseValidation"]["errors"][number];',
                directory, 'http_error_accepted_as_research_error',
            ),
        ]
    assert [r['exit_code'] for r in results] == [2, 2, 0], results
    output = {'finding': 'R21', 'results': results, 'network_database_or_provider_calls': 0}
    Path('/tmp/ra-review-wire-contracts.json').write_text(json.dumps(output, indent=2)+'\n')
    for result in results:
        print(result['case'], 'compiler exit', result['exit_code'])
        print(str(result['diagnostic'])[:600])


if __name__ == '__main__':
    main()
