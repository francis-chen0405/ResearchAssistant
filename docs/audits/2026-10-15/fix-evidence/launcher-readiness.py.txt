from __future__ import annotations

import json
import socket
import subprocess
import threading
import time
from pathlib import Path


def main() -> None:
    launcher = Path('Launch ResearchAssistant.command').read_text()
    assert '/usr/bin/curl --fail --silent "$WEB_URL"' in launcher
    assert 'for _ in {1..45}; do' in launcher
    stop = threading.Event()
    accepted = threading.Event()
    with socket.socket() as server:
        server.bind(('127.0.0.1', 0))
        server.listen(1)
        server.settimeout(0.2)
        port = server.getsockname()[1]

        def stall() -> None:
            while not stop.is_set():
                try:
                    peer, _address = server.accept()
                except TimeoutError:
                    continue
                with peer:
                    accepted.set()
                    stop.wait(50)
                return

        worker = threading.Thread(target=stall)
        worker.start()
        started = time.monotonic()
        process = subprocess.Popen(
            ['/usr/bin/curl', '--fail', '--silent', f'http://127.0.0.1:{port}/'],
            stdout=subprocess.DEVNULL,
            stderr=subprocess.PIPE,
        )
        try:
            assert accepted.wait(2)
            try:
                process.communicate(timeout=46)
            except subprocess.TimeoutExpired:
                observed = {
                    'launcher_web_readiness_options': ['--fail', '--silent'],
                    'nominal_loop_iterations': 45,
                    'first_request_completed': False,
                    'elapsed_seconds': round(time.monotonic() - started, 3),
                    'method': 'Exact launcher curl arguments against one owned loopback server that accepts but never responds. The full launcher, API, frontend and installed app are not started. The probe terminates/reaps only its own curl process and closes its own temporary socket.',
                }
            else:
                raise AssertionError('stalled request unexpectedly returned')
        finally:
            process.terminate()
            process.communicate(timeout=2)
            stop.set()
            worker.join(timeout=2)
            assert not worker.is_alive()
    output = json.dumps(observed, indent=2) + '\n'
    Path('/tmp/ra-review-launcher-readiness.json').write_text(output)
    print(output)


if __name__ == '__main__':
    main()
