from __future__ import annotations
import json,sys,tempfile
from collections import Counter
from datetime import UTC,datetime
from pathlib import Path
from typing import Any
from unittest.mock import patch
from uuid import uuid4
sys.path.insert(0,str(Path.cwd()/'tests'))
import test_phase9 as fixture
import researchassistant.research.historical_execution as history
from researchassistant.agents.supportingresearcher import _retrieve_result
from researchassistant.contracts.models import StatementReviewResult
from researchassistant.contracts.scraping import RetryPolicy,ScrapeRequest,ScrapeResponse
from researchassistant.storage.store import read_planner_output,read_statement_review

class CountingScraper:
    def __init__(self,resolved_url:str) -> None:
        self.resolved_url=resolved_url
        self.calls=0
    def scrape(self,request:ScrapeRequest) -> ScrapeResponse:
        self.calls+=1
        return ScrapeResponse(resolved_url=self.resolved_url,original_url=request.url,content_type='text/html',text='Generated repeat capture.')

def main() -> None:
    observations={}
    with tempfile.TemporaryDirectory(prefix='ra-retained-runner-') as folder:
        root=Path(folder)
        complete=fixture._run(root/'complete')
        assert complete.status is fixture.ProviderRunStatus.RELEASED
        pair=complete.researcher_result
        assert pair is not None
        snapshot=pair.supporting.retrieval_batch.snapshots[0]
        query=read_planner_output(complete.db_path,complete.run_id).search_queries[0].model_copy(update={'query_id':uuid4()})
        aliases=[]
        for label,url in (('original',snapshot.original_url),('resolved',snapshot.source_url)):
            assert url is not None
            state=history._seed_mvp11_deduplication(pair)
            seeded=dict(state.resolved_by_original)
            scraper=CountingScraper(snapshot.source_url)
            outcome,_=_retrieve_result(complete.run_id,query,1,url,scraper,RetryPolicy(),lambda:fixture.NOW,state,None)
            aliases.append({'alias':label,'scrape_calls':scraper.calls,'outcome':outcome.scrape_status.value,'attempts_made':outcome.attempts_made,'seeded_resolved_aliases':len(seeded)})
        observations['seeded_deduplication']={'results':aliases,'method':'Actual retained deduplication seeding from its real completed generated researcher artifact, followed by ordinary retrieval of the same original/resolved source URL. No database modifications or external calls.'}
        outcomes=tuple(item for side in (pair.supporting,pair.opposing) for item in side.retrieval_batch.outcomes)
        evaluation=history._mvp11_governor_input(complete.db_path,complete.run_id,(pair,pair),complete.analysis_result,history.ProviderOrchestrationConfig(),lambda:fixture.NOW)
        observations['duplicate_rate']={'actual_status_counts':dict(Counter(item.scrape_status.value for item in outcomes)),'actual_duplicate_outcomes':sum(item.scrape_status.value in {'duplicate_url','duplicate_content'} for item in outcomes),'reported_duplicate_count':evaluation.round_two_duplicate_count,'reported_result_count':evaluation.round_two_result_count,'method':'Actual retained Governor fact derivation using fully typed generated researcher/analysis artifacts; no new requests, budget edits or database writes.'}
        database=root/'interrupted.sqlite3'
        run_id=uuid4()
        original=history.insert_statement_review
        captured:list[StatementReviewResult]=[]
        def interrupted(path:str,review:StatementReviewResult) -> None:
            original(path,review)
            captured.append(review)
            raise KeyboardInterrupt('generated interruption after native Reviewer commit')
        def run(model:fixture.FakeLLMProvider,search:fixture.FakeSearchProvider,scraper:fixture.FakeScraperProvider) -> fixture.ProviderPipelineResult:
            return history.run_provider_pipeline('The fixture policy improves student outcomes.',db_path=database,search_provider=search,scraper_provider=scraper,llm_provider=model,run_id=run_id,clock=lambda:datetime.now(UTC))
        with patch.object(history,'insert_statement_review',interrupted):
            try:
                run(fixture.FakeLLMProvider(),fixture.FakeSearchProvider(),fixture.FakeScraperProvider())
            except KeyboardInterrupt:
                pass
            else:
                raise AssertionError('interruption not reached')
        review=captured[0]
        saved=read_statement_review(database,run_id,review.statement_draft_id)
        model,search,scraper=fixture.FakeLLMProvider(),fixture.FakeSearchProvider(),fixture.FakeScraperProvider()
        resumed=run(model,search,scraper)
        observations['reviewer_prefix']={'status':resumed.status.value,'failure_reason':resumed.failure_reason,'new_model_calls':len(model.requests),'new_search_calls':len(search.requests),'new_scrape_calls':len(scraper.requests),'original_review_unchanged':read_statement_review(database,run_id,review.statement_draft_id)==saved,'method':'Actual retained provider runner with advancing UTC clock, interrupted immediately after native Reviewer row commit; exact-source resume with no saved-artifact edits. All providers generated offline.'}
    Path('/tmp/ra-review-retained-runner-cases.json').write_text(json.dumps(observations,indent=2)+'\n')
    print(json.dumps(observations,indent=2))
if __name__=='__main__':
    main()
